<script>

  const canvas = document.getElementById('simCanvas');

  const ctx = canvas.getContext('2d');

  

  let currentRound = 1;

  let score = 0;

  let state = 'GUESSING';

  let roundSequence = [];

  let currentScenario = { person: 1, floor: 0 };

  

  let userPerson = 0;

  let userFloor = 0;

  

  let ballX = 0;

  let ballY = 0;

  let ballVy = 4;

  const gravity = 0.25;

  const floorY = 260;

  let trail = [];

  let speedParticles = [];


  function initScenarios() {

    const allScenarios = [];

    for (let p of [-1, 0, 1]) {

      for (let f of [-1, 0, 1]) {

        allScenarios.push({ person: p, floor: f });

      }

    }

    const fixedR1 = { person: 1, floor: 0 };

    const remaining = allScenarios.filter(s => !(s.person === fixedR1.person && s.floor === fixedR1.floor));

    

    for (let i = remaining.length - 1; i > 0; i--) {

      const j = Math.floor(Math.random() * (i + 1));

      [remaining[i], remaining[j]] = [remaining[j], remaining[i]];

    }

    

    roundSequence = [fixedR1, ...remaining];

  }


  function setupCanvas() {

    canvas.width = canvas.parentElement.clientWidth;

    canvas.height = canvas.parentElement.clientHeight;

  }

  window.addEventListener('resize', setupCanvas);

  setupCanvas();


  function loadRound(r) {

    currentRound = r;

    state = 'GUESSING';

    currentScenario = roundSequence[r - 1];

    

    ballX = canvas.width / 2;

    ballY = 100;

    ballVy = 2;

    trail = [];

    speedParticles = [];

    

    document.getElementById('roundDisplay').textContent = `Round ${currentRound} / 9`;

    document.getElementById('scoreDisplay').textContent = `Score: ${score}`;

    document.getElementById('actionBtn').textContent = "Check Prediction 💡";

    document.getElementById('resultsCard').classList.remove('visible');

    document.getElementById('speedHintBox').style.display = 'none';

    document.getElementById('controlsGrid').style.opacity = '1';

    document.getElementById('controlsGrid').style.pointerEvents = 'auto';

  }


  function getValidCombinations(targetNet) {

    const valid = [];

    for (let p of [-1, 0, 1]) {

      for (let f of [-1, 0, 1]) {

        if (p + f === targetNet) {

          valid.push({ person: p, floor: f });

        }

      }

    }

    return valid;

  }


  function formatCombo(c) {

    const pStr = c.person === -1 ? 'Left ⬅️' : c.person === 1 ? 'Right ➡️' : 'Still 🛑';

    const fStr = c.floor === -1 ? 'Left ⬅️' : c.floor === 1 ? 'Right ➡️' : 'Still 🛑';

    return `Person = ${pStr} | Floor = ${fStr}`;

  }


  document.querySelectorAll('.option-btn').forEach(btn => {

    btn.addEventListener('click', (e) => {

      const type = e.target.dataset.type;

      const val = parseInt(e.target.dataset.val);

      

      document.querySelectorAll(`.option-btn[data-type="${type}"]`).forEach(b => b.classList.remove('active'));

      e.target.classList.add('active');

      

      if (type === 'person') userPerson = val;

      if (type === 'floor') userFloor = val;

    });

  });


  document.getElementById('actionBtn').addEventListener('click', () => {

    if (state === 'GUESSING') {

      state = 'REVEALED';

      const targetNet = currentScenario.person + currentScenario.floor;

      const userNet = userPerson + userFloor;

      const isCorrect = targetNet === userNet;

      

      if (isCorrect) score++;

      document.getElementById('scoreDisplay').textContent = `Score: ${score}`;


      const badge = document.getElementById('statusBadge');

      const speedHint = document.getElementById('speedHintBox');


      if (isCorrect) {

        badge.className = "status-badge correct";

        badge.textContent = "✅ CORRECT! (Equivalent Motion)";

        speedHint.style.display = 'none';

      } else {

        badge.className = "status-badge incorrect";

        badge.textContent = "❌ INCORRECT";

        

        // Smart speed & direction hint updates (using "moving" instead of "drifting")

        if (targetNet !== 0 && Math.sign(targetNet) === Math.sign(userNet)) {

          if (Math.abs(targetNet) > Math.abs(userNet)) {

            speedHint.textContent = "💡 Hint: You got the direction right! But notice the ball is moving FASTER than what you picked (both frames are moving together).";

          } else {

            speedHint.textContent = "💡 Hint: You got the direction right! But notice the ball is moving SLOWER than what you picked (only one frame is moving).";

          }

          speedHint.style.display = 'block';

        } else if (targetNet === 0 && userNet !== 0) {

          speedHint.textContent = "💡 Hint: The ball is bouncing straight up and down! Look for a setup where the two motions cancel each other out.";

          speedHint.style.display = 'block';

        } else {

          speedHint.textContent = "💡 Hint: Check the direction! The ball is moving the opposite way.";

          speedHint.style.display = 'block';

        }

      }


      const validListEl = document.getElementById('validList');

      validListEl.innerHTML = '';

      

      const valids = getValidCombinations(targetNet);

      valids.forEach(c => {

        const li = document.createElement('li');

        const isUserChoice = (c.person === userPerson && c.floor === userFloor);

        li.className = `valid-item ${isUserChoice ? 'user-selected' : ''}`;

        li.textContent = `• ${formatCombo(c)} ${isUserChoice ? ' (Your Choice)' : ''}`;

        validListEl.appendChild(li);

      });


      document.getElementById('resultsCard').classList.add('visible');

      document.getElementById('controlsGrid').style.opacity = '0.5';

      document.getElementById('controlsGrid').style.pointerEvents = 'none';

      document.getElementById('actionBtn').textContent = currentRound < 9 ? "Next Round ➡️" : "Restart Game 🔄";

    } else {

      if (currentRound < 9) {

        loadRound(currentRound + 1);

      } else {

        score = 0;

        initScenarios();

        loadRound(1);

      }

    }

  });


  document.getElementById('resetBtn').addEventListener('click', () => {

    score = 0;

    initScenarios();

    loadRound(1);

  });


  function animate() {

    const netSpeed = currentScenario.person + currentScenario.floor;

    const speedMagnitude = Math.abs(netSpeed);


    ballX += netSpeed * 1.6;

    

    if (ballX > canvas.width + 20) ballX = -20;

    if (ballX < -20) ballX = canvas.width + 20;


    ballVy += gravity;

    ballY += ballVy;


    if (ballY >= floorY - 12) {

      ballY = floorY - 12;

      ballVy = -8.5;

    }


    const maxTrailLength = speedMagnitude === 2 ? 40 : speedMagnitude === 1 ? 20 : 8;

    trail.push({ x: ballX, y: ballY });

    if (trail.length > maxTrailLength) trail.shift();


    if (speedMagnitude === 2 && Math.random() < 0.3) {

      speedParticles.push({

        x: ballX - (netSpeed * 15),

        y: ballY + (Math.random() * 20 - 10),

        length: 25,

        alpha: 0.8

      });

    }


    ctx.clearRect(0, 0, canvas.width, canvas.height);


    for (let i = speedParticles.length - 1; i >= 0; i--) {

      const p = speedParticles[i];

      ctx.strokeStyle = `rgba(56, 189, 248, ${p.alpha})`;

      ctx.lineWidth = 1.5;

      ctx.beginPath();

      ctx.moveTo(p.x, p.y);

      ctx.lineTo(p.x - (netSpeed * p.length), p.y);

      ctx.stroke();

      p.alpha -= 0.04;

      if (p.alpha <= 0) speedParticles.splice(i, 1);

    }


    if (state === 'REVEALED') {

      ctx.strokeStyle = '#334155';

      ctx.lineWidth = 4;

      ctx.beginPath();

      ctx.moveTo(0, floorY);

      ctx.lineTo(canvas.width, floorY);

      ctx.stroke();


      const pX = ballX;

      ctx.strokeStyle = '#f8fafc';

      ctx.lineWidth = 3;

      ctx.beginPath();

      ctx.arc(pX, floorY - 40, 10, 0, Math.PI * 2);

      ctx.stroke();

      ctx.beginPath();

      ctx.moveTo(pX, floorY - 30); ctx.lineTo(pX, floorY - 15);

      ctx.moveTo(pX, floorY - 25); ctx.lineTo(pX - 10, floorY - 15);

      ctx.moveTo(pX, floorY - 25); ctx.lineTo(pX + 10, floorY - 15);

      ctx.moveTo(pX, floorY - 15); ctx.lineTo(pX - 8, floorY);

      ctx.moveTo(pX, floorY - 15); ctx.lineTo(pX + 8, floorY);

      ctx.stroke();

    } else {

      ctx.fillStyle = '#0f172a';

      ctx.fillRect(0, floorY, canvas.width, canvas.height - floorY);

    }


    const trailColor = speedMagnitude === 2 ? 'rgba(56, 189, 248, 0.7)' : 'rgba(245, 158, 11, 0.4)';

    ctx.strokeStyle = trailColor;

    ctx.lineWidth = speedMagnitude === 2 ? 3 : 2;

    ctx.beginPath();

    for (let i = 0; i < trail.length; i++) {

      if (i === 0) ctx.moveTo(trail[i].x, trail[i].y);

      else ctx.lineTo(trail[i].x, trail[i].y);

    }

    ctx.stroke();


    ctx.fillStyle = speedMagnitude === 2 ? '#38bdf8' : '#f59e0b';

    ctx.beginPath();

    ctx.arc(ballX, ballY, 12, 0, Math.PI * 2);

    ctx.fill();


    requestAnimationFrame(animate);

  }


  initScenarios();

  loadRound(1);

  requestAnimationFrame(animate);

</script>
